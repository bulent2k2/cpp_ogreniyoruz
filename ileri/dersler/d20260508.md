#include <iostream>
int main() {
    using S = unsigned;
    S n, toplam, b[101];
    std::cin >> n >> toplam;
    for (S k=1; k<=n; ++k) std::cin >> b[k];
    S f[(S)1e6+1][101];
    f[0][0] = 1;
    for (S k=1; k<=n; ++k) f[0][k] = 1;
    for (S x=1; x<=toplam; ++x)
        for (S k=1; k<=n; ++k) {
            S para = b[k];
            f[x][k] = f[x][k-1];
            if (para > x) continue;
            f[x][k] += f[x-para][k];
            f[x][k] %= (S)1e9+7;
        }
    std::cout << f[toplam][n] << std::endl;
    return 0;
}
