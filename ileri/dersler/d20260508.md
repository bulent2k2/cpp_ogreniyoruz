#include <iostream>
int main() {
    int n, toplam, b[101]; // bozuk paralar
    std::cin >> n >> toplam;
    for(int k=0; k<n; ++k)
        std::cin >> b[k];
    int f[toplam+1];
    f[0] = 0;
    for(int x=1; x<=toplam; ++x) {
       f[x] = toplam+1; // burda sonsuz yerine kullanabiliriz toplamı
       for(int k=0; k<n; ++k) // her bozuk para için
           if (b[k] <= x)
               f[x] = std::min(f[x], 1 + f[x-b[k]]);
    }
    std::cout << (f[toplam] <= toplam ? f[toplam] : -1) << std::endl;
    return 0;
}

    for (S k=1; k<=n; ++k) 
        std::cin >> b[k];
    f[0] = 1;
    for (S x=1; x<=toplam; ++x) {
        f[x] = 0;
        for (S k=1; k<=n; ++k) {
            S para = b[k];
            if (para > x) continue;
            f[x] += f[x-para];
            f[x] %= (S)1e9+7;
        }
    }
    std::cout << f[toplam] << std::endl;
    return 0;
}
